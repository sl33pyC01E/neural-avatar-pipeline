cmake_minimum_required(VERSION 3.22)
project(cleopatra_whisperx LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CT2_SOURCE}" ct2)
set_target_properties(ctranslate2 PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(cleo_whisperx SHARED bridge.cpp)
target_link_libraries(cleo_whisperx PRIVATE ctranslate2)
target_link_options(cleo_whisperx PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
